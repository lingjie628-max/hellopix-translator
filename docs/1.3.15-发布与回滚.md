```sql
BEGIN;
UPDATE client_releases SET enabled=false, updated_at=NOW() WHERE version='1.3.15';
UPDATE client_releases SET enabled=true,  updated_at=NOW() WHERE version='1.3.14';
UPDATE app_settings
SET value='"https://pixfanyi.com/downloads/Hello%20Pix-1.3.14-arm64.dmg"'::jsonb,
    updated_at=NOW()
WHERE key='downloads.client_url';
COMMIT;
```
